\section{Addition Formulas}
It is tempting to think
\[
  \cos(\alpha + \beta) = \cos\alpha + \cos\beta
\]
and that
\[
 \sin(\alpha + \beta) = \sin\alpha + \sin\beta .
\]
\begin{question} \label{Qffee}
Show both of these equations are false by choosing \emph{specific} values for $\alpha$ and $\beta$. Make simple choices.
\begin{freeResponse}
\end{freeResponse}
\end{question}

We now use vectors to find the correct expressions for $\cos(\alpha + \beta)$ and $ \sin(\alpha + \beta)$.

\begin{example} \label{Qd2fr4fds3}


Let points $A$, $B$ on the unit circle have respective polar angles $\alpha$, $\alpha + \beta$ as shown below.

\begin{enumerate}
\item Then $A$ has coordinates (type \emph{alpha} for $\alpha$, \emph{beta} for $\beta$)
\[
  (x_A, y_A) = (\answer{\cos \alpha}, \answer{\sin\alpha} )
\]
and $B$ coordinates
\[
 (x_B, y_B) = (\answer{\cos(\alpha+\beta)}, \answer{\sin(\alpha+\beta)} ).
\]

\item Now rotate point $A$ counterclockwise through the angle $\pi/2$ radians to get point $C$ with coordinates
\[
   (x_C, y_C) = (\answer{-\sin \alpha}, \answer{\cos\alpha} ).
\]

\item Since $\overrightarrow{OB}$ makes the angle $\beta$ with $\overrightarrow{OA}$, 
\begin{align*}
    \overrightarrow{OB} &= \overrightarrow{OA} \, \answer{\cos\beta} + \overrightarrow{OC} \, \answer{\sin\beta} \\
                                  &=\langle \cos\alpha, \sin\alpha\rangle \,  \answer{\cos\beta} + \langle -\sin\alpha, \cos\alpha \rangle \, \answer{\sin\beta}\\
                                  &= \langle  \answer{\cos\alpha \cos\beta - \sin\alpha \sin \beta} , \answer{\sin \alpha \cos\beta + \cos\alpha \sin\beta} \rangle .
\end{align*}

\item Then since 
\[
  \overrightarrow{OB} = \langle \cos(\alpha+\beta), \sin(\alpha+\beta)  \rangle,
\]
our conclusion is that
\[
  \cos(\alpha + \beta) = \cos\alpha \cos\beta - \sin\alpha \sin \beta
\]
and
\[
  \sin(\alpha + \beta) = \sin \alpha \cos\beta + \cos\alpha \sin\beta .
\]
\end{enumerate}
\end{example}

\section{Application: The Law of Sines}
\begin{example} \label{Exmv301zgjrg}
In $\Delta ABC$, angle $\angle A$ has measure $\alpha = \arctan(1/3)$ radians, angle $\angle B$ has measure $\beta = \arccos(-1/3)$ radians, and side $\overline{AB}$ has length $c =10$cm. 

\begin{enumerate}
\item Draw a reasonably accurate picture of $\Delta ABC$. 

\item Determine the exact lengths $b=AC$ and $a=BC$ of respective sides $\overline{AC}$ and  $\overline{BC}$. Your final expressions for these lengths should not have any trig functions. 
\end{enumerate}

\begin{explanation}
\begin{enumerate}
\item I'll leave this for you. But remember the definition of the inverse tangent and inverse cosine functions and what it means to measure an angle in radians.

\item By the law of sines with $\gamma$ the meaure of angle $\angle C$,
\[
  \frac{c}{\sin \gamma} = \frac{a}{\sin \alpha} = \frac{b}{\sin \beta}.
\]
Since 
\[
   \sin \gamma = \sin(\pi - (\alpha+\beta)) = \sin(\alpha+\beta),
\]
we have
\[
 \frac{10}{\sin(\alpha+\beta)} = \frac{a}{\sin \alpha} = \frac{b}{\sin\beta}.
\]

First we'll determine the values of
\[
  \sin \alpha = \sin (\arctan(1/3)),
\]
\[
  \sin \beta = \sin (\arccos(-1/3)),
\]
and
\begin{align*}
  \sin (\alpha+ \beta) &= \sin (\arctan(1/3) + \arccos(-1/3)) \\
                  &= \sin (\arctan(1/3))\cos(\arccos(-1/3)) + \cos (\arctan(1/3))\sin(\arccos(-1/3)),
\end{align*}
where the last equality uses the addition formula for $\sin(\alpha + \beta)$, namely
\[
  \sin(\alpha + \beta) = \sin \alpha \cos\beta + \cos\alpha \sin\beta .
\]

We'll do the first one together and I'll leave the rest for you.

The expression
\[
 \sin \alpha = \sin (\arctan(1/3))
\]
is the sine of the angle $\alpha = \arctan(1/3)$ between $0$ and $\pi/2$ whose tangent is $1/3$. 

Since
\[
  \tan\alpha = \frac{y}{x} = \frac{1}{3}
\]
and $0< \alpha < \pi/2$, we can choose $y=1$ and $x=3$ for the coordinates of a point $P$ with polar angle $\alpha$. Then the distance from $P$ to the origin is
\[
 r = \sqrt{x^2+y^2} = \sqrt{10}
\]
and
\[
   \sin\alpha = \frac{y}{r} = \frac{1}{\sqrt{10}}.
\]
\end{enumerate}
\end{explanation}
\end{example}
